function getWeather($city = 'Manila') {
    $apiKey = '41ecd6a02d67d4436012dc388c11db70';
    $url = "https://api.openweathermap.org/data/2.5/weather?q=" . urlencode($city) . "&units=metric&appid={$apiKey}";
    $data = file_get_contents($url); // Remove '@'
    return $data ? json_decode($data, true) : null;
}

function getNews() {
    $apiKey = '9iv2nGt6VR9XJ0_I6d-jnZN0Z_NjSOTXGW8nY1Kz0YrdMf0Q';
    $url = "https://api.currentsapi.services/v1/latest-news?language=en&apiKey={$apiKey}";
    $data = file_get_contents($url); // Removed '@' to catch errors

    if ($data) {
        $arr = json_decode($data, true);
        if (isset($arr['news'])) {
            return $arr['news']; // Return only the news list
        } else {
            echo "<pre style='color:red;'>Currents API Error:\n" . print_r($arr, true) . "</pre>";
        }
    }
    return null;
}



function getQuote() {
    $url = "https://zenquotes.io/api/today";
    $data = @file_get_contents($url);
    if ($data) {
        $arr = json_decode($data, true);
        return $arr[0] ?? null;
    }
    return null;
}

<script>
function openLightbox(src) {
    document.getElementById('lightbox-image').src = src;
    document.getElementById('lightbox-modal').classList.remove('hidden');
    document.getElementById('lightbox-modal').classList.add('flex');
}
function closeLightbox() {
    document.getElementById('lightbox-modal').classList.add('hidden');
    document.getElementById('lightbox-modal').classList.remove('flex');
    document.getElementById('lightbox-image').src = '';
}
function reactTo(id, emoji) {
    fetch('react.php', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ announcement_id: id, emoji: emoji })
    })
    .then(res => res.json())
    .then(data => {
        if (data.success) {
            const container = document.querySelector(`#emoji-counts-${id}`);
            if (container) {
                container.innerHTML = '';
                Object.entries(data.reactions).forEach(([emo, count]) => {
                    const span = document.createElement('span');
                    span.className = 'flex items-center gap-1 px-2 py-1 bg-gray-100 rounded-full';
                    span.innerHTML = `<span class="text-xl">${emo}</span><span class="text-sm">${count}</span>`;
                    container.appendChild(span);
                });
            }
            const btn = document.querySelector(`#react-btn-${id}`);
            if (btn) {
                btn.innerHTML = `${data.user_reaction || 'Like'}`;
            }
        }
    });
}
let hideTimeout;
function showReactions(container) {
    clearTimeout(hideTimeout);
    const menu = container.querySelector('.reaction-menu');
    menu.classList.remove('hidden');
    menu.classList.add('flex');
}
function hideReactions(container) {
    hideTimeout = setTimeout(() => {
        const menu = container.querySelector('.reaction-menu');
        menu.classList.add('hidden');
        menu.classList.remove('flex');
    }, 300);
}
