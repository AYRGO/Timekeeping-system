                <script>
                // Helper function for HTML escaping
                function h(str) {
                    return String(str).replace(/[&<>"']/g, function(match) {
                        return {'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'}[match];
                    });
                }

                // Initialize admin calendar on page load
                document.addEventListener('DOMContentLoaded', function() {
                    changeAdminMonth('<?= $ym ?>');
                });

                // AJAX function to change calendar month
                function changeAdminMonth(ym) {
                    const calendarGrid = document.getElementById('admin-calendar-grid');
                    const calendarTitle = document.getElementById('admin-calendar-title');
                    
                    calendarGrid.innerHTML = '<div class="col-span-7 text-center py-8"><i class="fas fa-spinner fa-spin text-2xl text-gray-400"></i><br><span class="text-gray-500 mt-2">Loading...</span></div>';
                    
                    fetch(`employee_calendar_admin_ajax.php?employee_id=<?= $employeeId ?>&ym=${ym}`)
                        .then(response => response.json())
                        .then(data => {
                            if (data.success) {
                                calendarGrid.innerHTML = data.calendar_html;
                                calendarTitle.textContent = data.month_title;
                                
                                // Update browser URL without refresh
                                const newUrl = new URL(window.location);
                                newUrl.searchParams.set('ym', ym);
                                window.history.pushState({}, '', newUrl);
                            } else {
                                throw new Error(data.error || 'Failed to load calendar');
                            }
                        })
                        .catch(error => {
                            console.error('Error loading calendar:', error);
                            calendarGrid.innerHTML = '<div class="col-span-7 text-center py-8 text-red-500"><i class="fas fa-exclamation-triangle text-2xl"></i><br><span class="mt-2">Error loading calendar. Please refresh the page.</span></div>';
                        });
                }

                // Schedule assignment modal functions

                    document.getElementById('scheduleAssignModal').classList.add('hidden');
                }

                function toggleScheduleFields() {
                    const assignmentType = document.getElementById('assignment_type').value;
                    const fixedField = document.getElementById('fixed_schedule_field');
                    const patternField = document.getElementById('pattern_schedule_field');
                    
                    if (assignmentType === 'pattern') {
                        fixedField.style.display = 'none';
                        patternField.style.display = 'block';
                    } else {
                        fixedField.style.display = 'block';
                        patternField.style.display = 'none';
                    }
                }

                // Admin schedule detail modal
                function openAdminScheduleModal(date, employeeId) {
                    // Show loading state

                    document.getElementById('adminScheduleModal').classList.remove('hidden');
                    
                    // Build detailed view for the selected date
                    const detailsHtml = `
                        <div class="space-y-4">
                            <div class="border-b pb-2 mb-4">
                                <h4 class="font-semibold text-lg">${new Date(date).toLocaleDateString('en-US', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' })}</h4>
                            </div>
                            <div class="text-center text-gray-500">
                                <p>Schedule details for this date will be implemented with additional features.</p>
                                <p class="mt-2 text-sm">Click elsewhere to close this modal.</p>
                            </div>
                        </div>
                    `;
                    
                    setTimeout(() => {
                        document.getElementById('admin-schedule-details').innerHTML = detailsHtml;
                    }, 500);
                }

                function closeAdminScheduleModal() {
                    document.getElementById('adminScheduleModal').classList.add('hidden');
                }

                // Approve/Reject override
                function approveOverride(overrideId, approve) {
                    const formData = new FormData();
                    formData.append('admin_action', 'approve_override');
                    formData.append('override_id', overrideId);
                    formData.append('approve', approve);

                    fetch('employee-edit.php?id=<?= $employeeId ?>', {
                        method: 'POST',
                        body: formData
                    })
                    .then(response => response.json())
                    .then(data => {
                        if (data.success) {
                            location.reload(); // Refresh to show updated status
                        } else {
                            alert('Error: ' + data.message);
                        }
                    })
                    .catch(error => {
                        console.error('Error:', error);
                        alert('An error occurred while processing the request');
                    });
                }

                    changeAdminMonth(ym);
