Detection Code

PHP Backend:

// Determine if this is a rest day request
$is_rest_day = empty($work_schedule_id);

// Check in Recent Activity
$isRestDay = (empty($activity['work_schedule_id']) || 
              ($activity['is_rest_day'] ?? 0) == 1);

echo $isRestDay ? 'Add a Day Off' : 'Schedule Change Request';

JavaScript Form:

// Handle rest day selection
if (scheduleId === 'rest_day') {
    document.getElementById('work_schedule_id').value = ''; // Empty
    console.log('Rest day selected');
} else {
    document.getElementById('work_schedule_id').value = scheduleId;
    console.log('Schedule change selected');
}

SQL Queries:

-- Get all rest day requests
SELECT * FROM schedule_change_requests 
WHERE is_rest_day = 1 OR work_schedule_id IS NULL;

-- Get all schedule change requests
SELECT * FROM schedule_change_requests 
WHERE is_rest_day = 0 AND work_schedule_id IS NOT NULL;
